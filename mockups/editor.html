# EMA trend with an RSI filter
from tradebot.indicators import indicator, ema, rsi

@indicator(name="EMA + RSI filter", params={"fast": 9, "slow": 21, "rsi_max": 70})
def signal(ohlc, fast, slow, rsi_max):
    f = ema(ohlc.close, fast)
    s = ema(ohlc.close, slow)
    r = rsi(ohlc.close, 14)
    long = (f > s) & (r < rsi_max)
        exit_ = (f < s) | (r > 85)   # rsi_exit is not defined
    return long, exit_
